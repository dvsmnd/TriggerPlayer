Raw serial input

Example Arduino sketch 
void setup() { Serial.begin(115200); }
void loop() {
  Serial.print(analogRead(A0));
  Serial.print(",");
  Serial.print(analogRead(A1));
  Serial.print(",");
  Serial.println(analogRead(A2));
  delay(20);
}
